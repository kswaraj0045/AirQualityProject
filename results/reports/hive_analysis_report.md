## 14 Yoy Changes
```sql
WITH yearly AS (SELECT record_year, ROUND(AVG(index_value), 2) AS avg_aqi FROM air_quality_records GROUP BY record_year) SELECT record_year, avg_aqi, LAG(avg_aqi, 1) OVER (ORDER BY record_year) AS prev_year_aqi, ROUND(avg_aqi - LAG(avg_aqi, 1) OVER (ORDER BY record_year), 2) AS aqi_change, ROUND((avg_aqi - LAG(avg_aqi, 1) OVER (ORDER BY record_year)) * 100.0 / LAG(avg_aqi, 1) OVER (ORDER BY record_year), 2) AS pct_change FROM yearly ORDER BY record_year ASC;
```
